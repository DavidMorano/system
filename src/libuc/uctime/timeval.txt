     struct timeval {
             time_t       tv_sec ;   /* seconds since 1 Jan 1970 */
             suseconds_t  tv_usec ;  /* and microseconds */
     } ;


= addition

Addition is simple.

int timeval_add(mut TIMEVAL *dst,CTIMEVAL *src1,CTIMEVAL *src2) noex {
	int		rs = SR_FAULT ;
	if (dst && src1 && src2) ylikely {
	    rs = SR_OK ;
	    dst->tv_sec = src1->tv_sec + src2->tv_sec ;
	    dst->tv_usec = src1->tv_usec + src2->tv_usec ;
	    if (dst->tv_usec >= onemillion) {
	        dst->tv_usec -= onemillion ;
	        dst->tv_sec += 1 ;
	    } /* end */
	} /* end if (non-null) */
	return rs ;


= subtraction

Subtraction is more complicated.  What happens to the sign bit on
the two contained values?

Normal subtraction (as advocated by the gods):

int timeval_sub(mut TIMEVAL *dst,CTIMEVAL *src1,CTIMEVAL *src2) noex {
	int		rs = SR_FAULT ;
	if (dst && src1 && src2) ylikely {
	    rs = SR_OK ;
	    dst->tv_sec = src1->tv_sec - src2->tv_sec ;
	    dst->tv_usec = src1->tv_usec - src2->tv_usec ;
	    if (dst->tv_usec < 0) {
	        dst->tv_usec += onemillion ;
	        dst->tv_sec -= 1 ;
	    } /* end */
	} /* end if (non-null) */
	return rs ;
} /* end subroutine (timeval_sub) */
