+ the function member pointer (C-language style function pointer call)

struct example ;
typedef int (example::*ex_m)(int = 0) noex ;
struct example {
    ex_m	m ;
    int callone(int) noex ;
    int calltwo(int) noex ;
} ;

{{
    example	ex ;
    int		rs ;
    ex.m = &example::callone ;
    rs = ex.m(43) ;

    example	*exp = &ex ;
    rs = (exp->*(exp->m))(42) ;
    rs = exp->m(42) ;
 }}
